Select your integration, install Pictum, and render your first icon.

Icons use a lowercase `prefix:name` identifier, such as `lucide:circle-check`.

<View title="React" icon="https://pictum.dev/v1/icons/devicon:react.svg">
  <Steps>
    <Step title="Install Pictum">
      Install the React package.

      <CodeGroup>
        ```bash npm
        npm install @pictum/react
        ```

        ```bash pnpm
        pnpm add @pictum/react
        ```

        ```bash bun
        bun add @pictum/react
        ```

        ```bash yarn
        yarn add @pictum/react
        ```
      </CodeGroup>
    </Step>

    <Step title="Render your first icon">
      ```tsx
      import { Icon } from "@pictum/react";

      export function Example() {
        return (
          <Icon
            name="lucide:circle-check"
            aria-label="Completed"
            width={24}
            height={24}
          />
        );
      }
      ```
    </Step>
  </Steps>

  Read the complete [React integration](/integrations/react).
</View>

      </CodeGroup>
    </Step>

    <Step title="Render your first icon">
      ```js
      import { icon } from "pictum";

      const image = document.createElement("img");
      image.src = icon("lucide:circle-check").url;
      image.alt = "Completed";
      image.width = 24;
      image.height = 24;

      document.body.append(image);
      ```
    </Step>
  </Steps>

<View title="PHP" icon="https://pictum.dev/v1/icons/skill-icons:php-light.svg">
  <Steps>
    <Step title="Install Pictum">
      Install the PHP package and Guzzle.

      ```bash
      composer require pictum/pictum guzzlehttp/guzzle
      ```
    </Step>

    <Step title="Render your first icon">
      ```php
      <?php

      use Pictum\Pictum;

      require __DIR__ . '/vendor/autoload.php';

      $pictum = new Pictum();
      ?>
      <span role="img" aria-label="Completed">
        <?php echo $pictum->icon('lucide:circle-check')->svg(); ?>
      </span>
      ```
    </Step>
  </Steps>

  Read the complete [PHP integration](/integrations/php).
